阴影 ​

阴影效果。

一直显示阴影
hover 时才显示阴影

<script setup lang="ts">
  import Card from "../src/card.vue";
  /**
   * Demo: 阴影
   *
   * 阴影效果。
   */
</script>
<template>
  <div class="flex-x-center flex-col flex-wrap gap-4">
    <Card class="card-strip-primary card-shadow" strip> <template #header>一直显示阴影</template></Card>
    <Card class="card-strip-info hover:card-shadow" strip> <template #header>hover 时才显示阴影</template></Card>
  </div>
</template>

标识条 ​

设置strip属性，可以在头部的前面添加竖条。

类名：card-strip-(primary|info|success|warning|danger)，如：card-strip-primary。

<script setup lang="ts">
  import Card from "../src/card.vue";
  /**
   * Demo: 标识条
   *
   * 设置`strip`属性，可以在头部的前面添加竖条。
   *
   * 类名：`card-strip-(primary|info|success|warning|danger)`，如：`card-strip-primary`。
   */
</script>
<template>
  <div class="flex-x-center flex-col flex-wrap gap-4">
    <Card class="" title="标题" strip> </Card>
    <Card class="card-strip-primary" title="标题" strip> </Card>
    <Card class="card-strip-info" title="标题" strip> </Card>
    <Card class="card-strip-success" title="标题" strip> </Card>
    <Card class="card-strip-warning" title="标题" strip> </Card>
    <Card class="card-strip-danger" title="标题" strip> </Card>
  </div>
</template>
